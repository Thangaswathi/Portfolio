        // Modal functionality
        function showSkillDetails(skillType) {
            const modal = document.getElementById('skillModal');
            const title = document.getElementById('skillTitle');
            const content = document.getElementById('skillContent');
            
            const skill = skillData[skillType];
            if (skill) {
                title.textContent = skill.title;
                content.innerHTML = skill.content;
                modal.classList.add('active');
            }
        }

        function closeModal() {
            const modal = document.getElementById('skillModal');
            modal.classList.remove('active');
        }

        // Add click events to skill bubbles
        document.addEventListener('DOMContentLoaded', () => {
            const skillBubbles = document.querySelectorAll('.skill-bubble');
            skillBubbles.forEach(bubble => {
                bubble.addEventListener('click', () => {
                    const skillType = bubble.getAttribute('data-skill');
                    showSkillDetails(skillType);
                });
            });

            // Close modal when clicking outside
            document.getElementById('skillModal').addEventListener('click', (e) => {
                if (e.target.id === 'skillModal') {
                    closeModal();
                }
            });

            // Close modal with Escape key
            document.addEventListener('keydown', (e) => {
                if (e.key === 'Escape') {
                    closeModal();
                }
            });
        });

        // Add some interactive particles
        function createInteractiveParticles() {
            const canvas = document.querySelector('.skills-canvas');
            
            setInterval(() => {
                const particle = document.createElement('div');
                particle.style.position = 'absolute';
                particle.style.width = '4px';
                particle.style.height = '4px';
                particle.style.background = 'linear-gradient(45deg, #6366f1, #a855f7)';
                particle.style.borderRadius = '50%';
                particle.style.pointerEvents = 'none';
                particle.style.left = Math.random() * 100 + '%';
                particle.style.top = '100%';
                particle.style.opacity = '0.6';
                
                canvas.appendChild(particle);
                
                // Animate particle
                let pos = 100;
                const animate = () => {
                    pos -= 0.5;
                    particle.style.top = pos + '%';
                    particle.style.opacity = (pos / 100) * 0.6;
                    
                    if (pos > -10) {
                        requestAnimationFrame(animate);
                    } else {
                        particle.remove();
                    }
                };
                animate();
            }, 2000);
        }

        // Initialize
        window.addEventListener('load', () => {
            createInteractiveParticles();
